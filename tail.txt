  if(r.cov < 15) return 'critical';
  if(r.cov < 30) return 'low';
  return 'healthy';
}
function healthLabel(cls){
  return {never_arrived:'Never Arrived', sold_out:'Sold Out', stagnant:'No Movement', critical:'Critical', low:'Low', healthy:'Healthy'}[cls] || cls;
}
function isNeverArrived(r){ return r.stock===0 && r.sold===0; }
function isSoldOut(r){ return r.stock===0 && r.sold>0; }
function isStagnant(r){ return r.stock>0 && r.sold===0; }
function covHealthFromDays(cov){
  if(cov < 15) return 'critical';
  if(cov < 30) return 'low';
  return 'healthy';
}

function invVal(r){ return r.stock * (r.price || 0); } // stock on hand valued at current price

function stPct(sold, stock){ return (sold+stock) ? sold/(sold+stock)*100 : 0; }

// Sell-through rise vs the 1 Mayâ€“18 Jul ("mid-July") baseline, tiered by how many percentage points it climbed.
// riseTier(null) means no baseline exists for that product (launched after mid-July) â€” never flagged, since there's nothing to compare against.
function riseTier(sold, stock, soldBase, stockBase){
  if(soldBase === null || soldBase === undefined) return null;
  const delta = stPct(sold, stock) - stPct(soldBase, stockBase);
  if(delta >= 15) return 'tier3';
  if(delta >= 10) return 'tier2';
  if(delta >= 5) return 'tier1';
  return null;
}
const RISE_TIER_LABEL = {tier1:'+5â€“10pt', tier2:'+10â€“15pt', tier3:'+15pt+'};

function aggregate(rows){
  const rev = rows.reduce((s,r)=>s+r.rev,0);
  const sold = rows.reduce((s,r)=>s+r.sold,0);
  const stock = rows.reduce((s,r)=>s+r.stock,0);
  const ads = rows.reduce((s,r)=>s+r.ads,0);
  const st = (sold+stock) ? sold/(sold+stock)*100 : 0;
  const avgVal = sold ? rev/sold : 0;
  const invValue = rows.reduce((s,r)=>s+invVal(r),0);
  const newCount = rows.filter(r=>r.isNew).length;
  const neverArrived = rows.filter(isNeverArrived).length;
  const stagnant = rows.filter(isStagnant).length;
  // "at risk" excludes never-arrived and stagnant SKUs: those aren't a coverage-speed problem
  const under30 = rows.filter(r=>{
    const cls = healthClass(r);
    return cls==='critical' || cls==='low' || cls==='sold_out';
  }).length;
  return {rev, sold, stock, ads, st, avgVal, invValue, newCount, under30, neverArrived, stagnant, count: rows.length};
}

function pctDelta(now, prev){
  if(prev === 0) return now===0 ? 0 : null; // null = undefined/new
  return (now-prev)/prev*100;
}

function deltaChip(now, prev, opts={}){
  const d = pctDelta(now, prev);
  if(d === null) return '<span class="delta flat">â€” new</span>';
  const dir = d > 0.05 ? 'up' : d < -0.05 ? 'down' : 'flat';
  const arrow = dir==='up' ? 'â–²' : dir==='down' ? 'â–¼' : 'â€¢';
  const sign = d>0 ? '+' : '';
  return `<span class="delta ${dir}">${arrow} ${sign}${d.toFixed(1)}%</span>`;
}

function buildCategoryMap(rows){
  const map = new Map();
  for(const r of rows){
    if(!map.has(r.c)) map.set(r.c, []);
    map.get(r.c).push(r);
  }
  return map;
}

function applyFilters(rows){
  return rows.filter(r=>{
    if(state.search && !r.n.toLowerCase().includes(state.search.toLowerCase())) return false;
    if(state.category!=='all' && r.c!==state.category) return false;
    if(state.health!=='all' && healthClass(r)!==state.health) return false;
    if(r.rev < state.minRev) return false;
    if(state.notesFilter!=='all'){
      const note = state.notes.get(r.n) || '';
      if(state.notesFilter==='none' ? note!=='' : note!==state.notesFilter) return false;
    }
    return true;
  });
}

function sortRows(rows, override){
  const sortBy = override ? override.by : state.sortBy;
  const dir = (override ? override.dir : state.sortDir) === 'asc' ? 1 : -1;
  const key = {rev:'rev', name:'n', stock:'stock', coverage:'cov', sellthrough:null, invvalue:null, isnew:null}[sortBy];
  return [...rows].sort((a,b)=>{
    if(sortBy==='name') return dir*a.n.localeCompare(b.n);
    if(sortBy==='isnew'){
      const diff = (b.isNew?1:0)-(a.isNew?1:0); // New items first regardless of asc/desc toggle
      return diff !== 0 ? diff : (b.rev - a.rev); // within each group, higher revenue first
    }
    if(sortBy==='sellthrough'){
      const sa = a.sold/(a.sold+a.stock||1), sb = b.sold/(b.sold+b.stock||1);
      return dir*(sa-sb);
    }
    if(sortBy==='invvalue') return dir*(invVal(a)-invVal(b));
    return dir*(a[key]-b[key]);
  });
}

function topCategoriesByDailySales(rows, n=5){
  const map = buildCategoryMap(rows);
  const list = [...map.entries()].map(([cat, rs])=>({cat, ads: rs.reduce((s,r)=>s+r.ads,0)}));
  list.sort((a,b)=>b.ads-a.ads);
  return list.slice(0,n);
}

function findMatch(name, rows){ return rows.find(r=>r.n===name); }

      if(state.openCats.has(cat)) state.openCats.delete(cat); else state.openCats.add(cat);
      el.closest('.group').classList.toggle('open');
    });
  });

  wrap.querySelectorAll('[data-delcat]').forEach(btn=>{
    btn.addEventListener('click', e=>{
      e.stopPropagation(); // don't also trigger the group-head expand/collapse
      const cat = btn.dataset.delcat;
      if(confirm(`Remove the entire "${cat}" type from the dashboard? You can bring it back from the "hidden" bar above the filters.`)){
        state.deletedCats.add(cat);
        render();
      }
    });
  });

  const GROUP_SORT_DIR = { isnew:'desc', rev:'desc', stock:'desc', invvalue:'desc', sellthrough:'desc', coverage:'asc', name:'asc' };
  wrap.querySelectorAll('[data-groupsort]').forEach(sel=>{
    sel.addEventListener('click', e=>e.stopPropagation());
    sel.addEventListener('change', e=>{
      e.stopPropagation();
      const cat = sel.dataset.groupsort;
      const by = sel.value;
      if(by) state.groupSort.set(cat, {by, dir: GROUP_SORT_DIR[by]});
      else state.groupSort.delete(cat);
      state.openCats.add(cat); // opening the group so the new order is immediately visible
      render();
    });
  });

  wrap.querySelectorAll('[data-delsku]').forEach(btn=>{
    btn.addEventListener('click', e=>{
      e.stopPropagation();
      const name = btn.dataset.delsku;
      state.deletedSkus.add(name);
      render();
    });
  });

  wrap.querySelectorAll('[data-note-sku]').forEach(sel=>{
    sel.addEventListener('click', e=>e.stopPropagation());
    sel.addEventListener('change', e=>{
      e.stopPropagation();
      const name = sel.dataset.noteSku;
      const val = sel.value;
      if(val) state.notes.set(name, val); else state.notes.delete(name);
      render();
    });
  });

  attachHoverImages();
}

function escHtml(s){ return String(s).replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }

function attachHoverImages(){
  const tip = document.getElementById('imgTip');
  document.querySelectorAll('.hoverable').forEach(el=>{
    el.addEventListener('mouseenter', e=>{
      const img = el.dataset.img;
      const name = el.dataset.name;
      const src = img ? escAttr(img) : PLACEHOLDER_IMG;
      tip.innerHTML = `<img src="${src}" onerror="this.onerror=null;this.src='${PLACEHOLDER_IMG}'" alt=""><div class="cap">${escHtml(name)}</div>`;
      tip.style.display = 'block';
      positionTip(e);
    });
    el.addEventListener('mousemove', positionTip);
    el.addEventListener('mouseleave', ()=>{ tip.style.display='none'; });
  });
}
function positionTip(e){
  const tip = document.getElementById('imgTip');
  const pad = 16;
  let x = e.clientX + pad, y = e.clientY + pad;
  if(x + 184 > window.innerWidth) x = e.clientX - 184 - pad;
  if(y + 220 > window.innerHeight) y = e.clientY - 220;
  tip.style.left = x+'px';
  tip.style.top = y+'px';
}

function csvEscape(val){
  const s = String(val ?? '');
  if(/[",\n]/.test(s)) return '"' + s.replace(/"/g,'""') + '"';
  return s;
}

function exportFilteredCsv(){
  const period = getPeriod(state.periodId);

  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}

async function processWithConcurrency(items, limit, worker){
  let idx = 0;
  async function runOne(){
    while(idx < items.length){
      const i = idx++;
      await worker(items[i], i);
    }
  }
  await Promise.all(Array.from({length: Math.min(limit, items.length)}, runOne));
}

async function fetchImageAsResizedBase64(url, maxSize){
  const resp = await fetch(url, {mode:'cors'});
  if(!resp.ok) throw new Error('http '+resp.status);
  const blob = await resp.blob();
  const bitmap = await createImageBitmap(blob);
  const scale = Math.min(1, maxSize / Math.max(bitmap.width, bitmap.height));
  const w = Math.max(1, Math.round(bitmap.width * scale));
  const h = Math.max(1, Math.round(bitmap.height * scale));
  const canvas = document.createElement('canvas');
  canvas.width = w; canvas.height = h;
  canvas.getContext('2d').drawImage(bitmap, 0, 0, w, h);
  return { base64: canvas.toDataURL('image/jpeg', 0.85), width: w, height: h };
}

  xlsxBtn.textContent = 'â³ Preparingâ€¦';

  try{
    const wb = new ExcelJS.Workbook();
    const ws = wb.addWorksheet('Inventory Export');
    ws.columns = [
      {header:'Photo', key:'photo', width: PHOTO_COL_WIDTH},
      {header:'Product Name', key:'name', width:34},
      {header:'Category', key:'cat', width:22},
      {header:'Collected Revenue', key:'rev', width:16},
      {header:'Items Sold', key:'sold', width:11},
      {header:'Available Stock', key:'stock', width:13},
      {header:'Current Price', key:'price', width:12},
      {header:'Inventory Value', key:'invval', width:14},
      {header:'Avg Daily Sales', key:'ads', width:13},
      {header:'Stock Coverage (days)', key:'cov', width:17},
      {header:'Sell-Through %', key:'st', width:13},
      {header:'Stock Health', key:'health', width:13},
      {header:'New Collection', key:'isnew', width:13},
      {header:'Note', key:'note', width:14},
    ];
    ws.getRow(1).font = {bold:true};
    ws.getRow(1).alignment = {vertical:'middle'};

    rows.forEach((r, i)=>{
      const cls = healthClass(r);
      const st = r.sold/(r.sold+r.stock||1)*100;
      const note = state.notes.get(r.n) || '';
      ws.addRow({
        photo:'', name:r.n, cat:r.c, rev:r.rev, sold:r.sold, stock:r.stock,
        price: r.price || '', invval: r.price ? invVal(r) : '', ads:r.ads, cov:r.cov,
        st: Number(st.toFixed(1)), health: healthLabel(cls), isnew: r.isNew ? 'Yes' : 'No',
        note: note ? NOTE_LABELS[note] : ''
      });
      ws.getRow(i+2).height = PHOTO_ROW_HEIGHT;
    });

    let done = 0, failed = 0;
    await processWithConcurrency(rows, 6, async (r, i)=>{
      done++;
      status.textContent = `Fetching photosâ€¦ ${done}/${rows.length}`;
      if(!r.img) return;
      try{
        const {base64} = await fetchImageAsResizedBase64(r.img, PHOTO_PX);
        const ext = base64.startsWith('data:image/png') ? 'png' : 'jpeg';
        const imgId = wb.addImage({base64, extension: ext});
        ws.addImage(imgId, { tl:{col:0.15, row:(i+1)+0.1}, ext:{width:PHOTO_PX, height:PHOTO_PX} });
      } catch(e){
        failed++;
      }
    });

    status.textContent = 'Generating fileâ€¦';
    const buffer = await wb.xlsx.writeBuffer();
    const blob = new Blob([buffer], {type:'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    const dateStr = new Date().toISOString().slice(0,10);
    a.href = url;
    a.download = `dashboard-export-photos_${period.id}_${dateStr}.xlsx`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);

    status.textContent = failed>0
      ? `Done â€” ${failed} of ${rows.length} photo(s) couldn't be loaded (broken link or blocked by the browser) and were left blank.`
      : `Done â€” all ${rows.length} photos embedded.`;
  } catch(e){
    status.textContent = 'Export failed: ' + e.message;
  } finally {
    xlsxBtn.disabled = false; csvBtn.disabled = false;
    xlsxBtn.textContent = 'ðŸ–¼ï¸ Export Excel + Photos';
    setTimeout(()=>{ status.textContent=''; }, 8000);
  }
}
